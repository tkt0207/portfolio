概要
<canvas>はHTML上に置ける「白紙の画用紙」のようなタグで、JavaScriptで直接ピクセルを描くための要素です。
グラフィックやアニメーションを描画することができます。
使い方
HTMLでcanvasを定義し、JavaScriptでcanvasの中身を描画します。
HTML
<!-- canvasを定義 -->
<canvas width="200" height="200"></canvas>
JS
// canvasの描画コンテキストを取得
const canvas = document.querySelector("canvas");
const ctx = canvas.getContext("2d");

// canvasの横幅と高さを設定
ctx.width = 500;
ctx.height = 400;

/*======================================*/
/* 下記は基本的なキャンバス描画処理です */
/*======================================*/
//  1. キャンバスクリア
ctx.crearRect(0, 0, ctx.width, ctx.height);  // (x,y,width,height)

//  2. スタイル設定
ctx.fillStyle = "blue";         // 塗りつぶし色
ctx.strokeStyle = "#ff0000";    // 線の色
ctx.lineWidth = 4;              // 線の幅
ctx.lineCap = "butt";           // 線の末端の形 [butt, round, square]
ctx.lineJoin = "round";         // 線と線が交わる交点の形 [round, bebel, miter]
ctx.globalAlpha = 0.8;          // 透明度

//  3. 塗りつぶし長方形の描画
ctx.fillRect(10, 20, 100, 160);  // (x,y,width,height)

//  4. 長方形の枠を描画
storokeRect(10, 20, 100, 160);   // (x,y,width,height)

//  5. パスの作成
ctx.beginPath();                                  // パスの開始
ctx.moveTo(x, y);                                 // 指定の座標(x,y)にパスを移動
ctx.lineTo(x, y);                                 // 現在のパス位置から指定の座標(x,y)に線を引く
ctx.arcTo(x1, y1, x2, y2, radius);                // 現在のパス位置から円弧を引く
ctx.quadraticCurveTo(cp1x, cp1y, x, y);           // 現在の位置から指定座標(x,y)まで二次ベジェ曲線を引く
ctx.bezierCurveTo(cp1x, cp1y, cp2x, cp2y, x, y);  // 現在の位置から指定座標(x,y)まで三次ベジェ曲線を引く
ctx.rect(x, y, width, height);                    // (x,y)を左上として長方形の輪郭を引く
ctx.arc(x, y, radius, startAngle, endAngle, counterclockwise);  // (x,y)を中心に円の輪郭を引く
ctx.closePath();                                  // パスを閉じる(現在のパスを開始地点に繋げる)

//  6. パス内の塗りつぶし
ctx.fill();

//  7. パスの線付け
ctx.stroke();

//  8. 文字のスタイル設定
ctx.font = "bold 12px Arial";  // フォント設定
ctx.textAlign = "center";      // 横位置設定 [start, end, left, right, center]
ctx.textBaseline = "middle";   // 縦位置設定 [top, hanging, middle, alphabetic, ideographic, bottom]

//  9. 文字の描画
fillText("文字", 20, 40);  // (text,x,y[,naxWidth])

// 10. 文字の輪郭の描画
strokeText("文字の輪郭", 20, 40);  // (text,x,y[,naxWidth])

// 11. 画像の描画
drawImage(image, 0, 0, 200, 300);  // (text,x,y[,width,height])

// 12. アニメーション
let raf;
function func(){
    // 描画処理
    raf = requestAnimationFrame(func);    
}
raf = requestAnimationFrame(func);  // アニメーション開始
cancelAnimationFrame(raf);          // アニメーション停止

// XX. 他にも設定の保存と復元、グラデーションの作成、影つけ、座標変換、クリップ、合成方法の設定などがあります
// 説明が長くなるのでここには記載しません

概要
「display: flex;」は要素を一列（または一行）として並べ、配置や余白を柔軟に制御するためのレイアウト方式です。
要素を横に並べたり、中央に揃えたりすることが簡単にできます。
使い方
親要素で子要素の並び方や方向、余白を設定し、子要素で要素の伸び縮みを設定します。
HTML
<!-- flexコンテナを定義 -->
<div class="parent">
    <div class="child">1</div>
    <div class="child">2</div>
    <div class="child">3</div>
    <div class="child">4</div>
</div>
CSS
/* 親要素 */
.parent{
    /* displayをflexに設定 */
    display: flex;

    /* 主軸の向きを設定 */
    flex-direction: column;

    /* 主軸に収まらないときに改行するかを設定 */
    flex-wrap: wrap;

    /* 主軸の配置方法を設定 */
    justify-content: center;

    /* 交差軸の配置方法を設定 */
    align-items: center;

    /* 要素間の余白を設定 */
    gap: 10px;
}

/* 子要素 */
.child{
    /* 初期サイズ */
    flex-basis: 200px;

    /* 余白があるときの伸びる比率 */
    flex-grow: 1;

    /* 要素が収まらないときの縮む比率 */
    flex-shrink: 0;

    /* 配置順 */
    order: unset;
}

概要
「display: grid;」は要素を列×行で制御するためのレイアウト方式です。
要素の位置(1列目の2行目など)を設定したり、列結合や行結合なども可能です。
使い方
親要素で列(行)の数や幅を設定し、子要素で配置する場所や結合数を設定します。
HTML
<!-- gridコンテナを定義 -->
<div class="parent">
    <div class="child">1</div>
    <div class="child">2</div>
    <div class="child">3</div>
    <div class="child">4</div>
</div>
CSS
/* 親要素 */
.parent{
    /* displayをgridに設定 */
    display: grid;

    /* 列数と各幅を定義 */
    grid-template-columns: 200px 1fr 200px;
    /* よく使うのは下記の書き方です。1セルの最小値を240pxとしてセルを埋めれるだけ埋めます */
    grid-template-columns: repeat(auto-fill, minmax(240px, 1fr));

    /* 行数と幅を定義 */
    grid-template-rows: auto 1fr auto;

    /* セルに名前を割り当て */
    grid-template-areas:
       "a a a"
       "b b c"
       "b b c";

    /* 要素間の余白を設定 */
    gap: 10px;
}

/* 子要素 */
.child{
    /* 列位置を設定(数値は線基準です。spanの後は結合数になります) */
    grid-column: 1 / span 2;

    /* 行位置を設定(数値は線基準です。) */
    grid-row: 2 / 3;

    /* 配置位置をセル名から設定 */
    grid-area: a;
}

概要
@mediaは画面サイズや環境に応じてCSSを切り替える仕組みです。
画面幅が600px以下の時や、印刷時のみスタイルを当てたい際に重宝します。
使い方
任意の「メディア種別」と任意の数の「メディア特性」の式で構成します。
「論理演算子」を使用して、複数のクエリーを様々な形で組み合わせることができます。
CSS
/* フォーマット */
@media 【メディア種別】 【論理演算子】 (【メディア特性】)... {
    /* 適用したいスタイル */
}

/* 例 */
@media screan and (max-width: 1000px) {
    .box{
        padding: 8px 2px;
    }
}
種類
「メディア種別」「メディア特性」「論理演算子」にはそれぞれ下記の種類があります。
メディア種別
all すべての機器
print 印刷時
screan 画面
メディア特性
width ビューポートの幅
height ビューポートの高さ
aspect-ratio ビューポートのアスペクト比
orientation ビューポートの向き
prefers-color-scheme ライトモード/ダークモード
display-mode 表示モード。全画面やピクチャインピクチャ等
hover ホバーの使用可否(主要な入力装置で)
any-hover ホバーの使用可否(全ての入力装置で)
pointer ポインティングデバイスの有無(主要な入力装置で)
any-pointer ポインティングデバイスの有無(全ての入力装置で)
color 出力機器の色のビット数
他色々
論理演算子
and 結合させた各クエリーがtrueならtrue
or 結合させた各クエリーが、1つでもtrueならtrue
, orと同じ
not 直後のクエリーの論理を反転
only クエリー全体がtrueならtrue。現在は使わない

概要
@keyframesはCSSアニメーションの動きを定義する仕組みです。
どのタイミングでどの状態になるかを時間軸で定義できます。
使い方
アニメーション名を指定し、「from-to」もしくは「%」でキーフレームを設定します。
定義したアニメーションは、適用したいセレクタに「animation:」で設定します。
CSS
/* from-to */
@keyframes animation1 {
    from{
        filter: hue-rotate(0deg);
    }
    to{
        filter: hue-rotate(360deg);
    }
}

/* % */
@keyframes animation2 {
    0%,100%{
        opacity: 1;
    }
    50%{
        opacity: 0;
    }
}

/* アニメーションの適用 */
.box{
    /* [アニメーション名][実行時間][イージング][回数][実行前後のスタイル適用方法] */
    /* [遅延時間][方向][実行状態]なども設定可能 */
    animation: animation1 2s ease infinite both;
}

概要
カスタム変数(カスタムプロパティ)はCSSで使える変数です。
色を管理したり、JavaScriptや@mediaと組み合わせることで簡単に全体のスタイルを変えることもできます。
使い方
"--"で始まる変数を、任意のセレクタ内に定義します。
カスタム変数は定義したセレクタ(要素)自身とその子孫要素でのみ参照できます。
JavaScriptにおけるブロックスコープと似たようなイメージです。
CSS
/* :root内に定義することにより、document全体で参照できます。 */
:root{
    --color-back: #f0f0f0;
    --color-font: #444444;
}

.box{
    /* カスタム変数を参照するときは、var([変数名]) です。 */
    background-color: var(--color-back);

    /* .boxの中で定義した場合、.box自身かその子孫要素でのみ参照できます。 */
    --item-width: 200px;

    /* 再宣言することも出来ますが、変更後の値が反映されるのは自身と子孫要素のみです。 */
    --color-font: red;
}

.item{
    /* .itemが.boxの子孫要素の場合、200px が設定されます。 */
    width: var(--item-width);

    /* .itemが.boxの子孫要素の場合、red が設定されます。子孫要素でない場合は #444444 が設定されます。 */
    color: var(--color-font);
}

概要
疑似クラスは要素の状態や条件に応じてCSSを適用する仕組みです。
要素がhoverされてるときや、その要素が何番目かなどの条件に応じてスタイルを設定できます。
使い方
セレクタの後ろに":"で始まる疑似クラスを指定します。
CSS
/* 任意のセレクタの末尾に疑似クラスを指定します。 */
.box:hover{
    color: yellow;
}

/* 疑似クラスを重ねることも可能です。 */
.box:hover:active{
    color: red;
}

概要
疑似要素は選択された要素の特定の部分にスタイル付けできるようにする仕組みです。
特に::before(要素の前に追加)と::after(要素の後に追加)は、拡張性が高くかなり便利です。
使い方
セレクタの後ろに"::"で始まる疑似要素を指定します。
CSS
/* 任意のセレクタの末尾に疑似要素を指定します。 */
.box::before{
    /* ::bofore, ::afterの場合、contentプロパティの設定が必須です。 */
    content: "￥";

    /* 疑似要素に設定したいスタイルを記述 */
    color: red;
}

/* 疑似クラスの後に疑似要素を書くことも可能です。(逆は出来ません) */
.box:hover::placeholder{
    color: blue;
}

概要
import/exportを利用することで、JavaScriptのファイルをモジュール化できます。
外部に公開したいものだけをexportし、必要なものだけをimportすることで、カプセル化を実現できます。
使い方
① export側で公開したいシンボル(関数、定数、クラス)を指定。
② import側で使用したいシンボルを指定。
③ htmlでJavaScriptを読み込む際のtypeを "module" に設定。
export.js
// 個別にexport
export const PAGE_NUM = 4;
export function fix_date(){}

// まとめてexport
const PAGE_WIDTH = 300;
funtion calc_value(){}
export { PAGE_WIDTH, calc_value };

// デフォルトexport (1ファイルにつき1つ)
export default function ex(){}
import.js
// 個別にimport
import { PAGE_NUM, calc_value } from "./export.js";

// 名前を付けてimport
import { fix_date as fd } from "./export.js";

// デフォルトimport
import Ex from "./export.js";

// 全てimport
import * as ExModule from "./export.js";

/* 全てimportの場合は、付けた名前.シンボルで使用できます */
ExModule.calc_value();
index.html
<script type="module" src="./import.js"></script>
概要
fetchはブラウザからHTTP通信するための標準APIです。
特にSPAの実装では重宝し、ブラウザのリロードなしでサーバーと通信できます。
使い方
HTTPメソッドや送信データを指定してURLにリクエストを送信します。
非同期処理で動作するため、通信中でも画面を操作できます。
データはjson形式かFormData形式で送信することが多いです。他にも形式はあります。
JavaScript
// 送信データを指定
/* 送信データがjson形式の場合 */
const send_data = JSON.stringfy({
    name: "山田 太郎",
    id: "2395",
    bmn: "80"
})

/* 送信データがFormData形式の場合 */
const send_data = new FormData();
send_data.append("name", "山田太郎");
send_data.append("file", file_input.files[0]);


// fetch(URL, 送信データ)でリクエストを投げる
// 通信エラーを考慮し、try/catch 内で行うのが無難
try{
    const res = await fetch(url, {
        /* HTTPメソッドを指定 ※指定しない場合はGETになる */
        method: 'POST',

        /* ヘッダを指定 ※FormData形式で送信する場合は不要 */
        headers: {
            'Content-Type': 'application/json'
        },

        /* 送信データを指定 */
        body: send_data
    });
	
    if(!res.ok){
        // HTTPエラー(404/500)時の処理
    }

    // サーバーからレスポンスデータを取得
    /* レスポンスがjson形式の場合 */
    const data = await res.json();

    /* レスポンスがBlob形式の場合 */
    const data = await res.blob();

    // データ取得後の処理(UI更新など)

} catch(e){
    // 通信が失敗した時の処理
}
